# Python build requirements for RAFT on any CUDA 13.x toolkit (aarch64 DGX
# Spark / GB10, and x86_64 RTX 40xx / RTX 50xx). Renamed from
# requirements-build-cuda132.txt: nothing in here is pinned to a specific
# CUDA *minor* version -- gb10/rtx40xx/rtx50xx's raft_env_*.sh scripts all
# auto-detect and build against the highest installed toolkit (13.2, 13.3,
# ...), so a filename naming one specific minor version was misleading.
# "13x" (not bare "13") matches this repo's existing cupy-cuda13x convention
# for "any CUDA 13.x minor version".
#
# Install with:
#   pip install -r requirements-build-cuda13x.txt \
#     --extra-index-url https://pypi.anaconda.org/rapidsai-wheels-nightly/simple

--extra-index-url=https://pypi.anaconda.org/rapidsai-wheels-nightly/simple

# Build tools
# rapids-build-backend 0.4.1 is the latest in the 0.4.x series
rapids-build-backend>=0.4.1,<0.5.0
# scikit-build-core 0.12.x: 0.12.2 validated; cap at 0.13 to avoid breaking changes
scikit-build-core[pyproject]>=0.12.0,<0.13.0
# Cython (required to build pylibraft's and raft-dask's .pyx extensions).
# raft_wheel_*.sh use `pip wheel --no-build-isolation`, so pyproject.toml's
# own build-time "cython>=3.2.2" requirement is never auto-installed -- it
# has to be pre-installed here instead. Missing entirely until pylibraft's
# wheel was actually built for the first time (previously only libraft's
# wheel, which has no Cython extensions, had ever been exercised).
cython>=3.2.2

# CUDA 13.x Python bindings — 13.2.0 is the latest; cap at <14.0 for CUDA major ABI
cuda-python>=13.2.0,<14.0

# RAPIDS dependencies (nightly wheels, CUDA 13 suffixed, pinned to match
# this repo's current VERSION minor -- must track VERSION/dependencies.yaml's
# depends_on_librmm pin (26.8.* as of 2026-07-07) or pip's resolver rejects
# the built libraft-gb10/pylibraft-gb10 wheels as conflicting with this pin.
librmm-cu13==26.8.*,>=0.0.0a0
rmm-cu13==26.8.*,>=0.0.0a0
rapids-logger==0.2.*,>=0.0.0a0

# UCX (required for raft-dask); 1.19.0 is the only 1.19 release available
libucx-cu13>=1.19.0,<1.19.1

# NCCL (required for raft-dask's comms_utils/nccl Cython extensions,
# depends_on_nccl in dependencies.yaml) -- was missing from this file
# entirely until raft-dask was added to the variant wheel pipeline, since
# neither libraft nor pylibraft need it.
nvidia-nccl-cu13>=2.19
